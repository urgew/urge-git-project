5. Create the cd.yml file
----------------------------
Create .github/workflows/cd.yml in your repository and paste this configuration

name: Continuous Deployment via AWX

on:
  push:
    branches:
      - main

jobs:
  deploy-via-awx:
    runs-on: self-hosted

    steps:

      # ============================================================
      # Step 1: Check GitHub Secrets
      # ============================================================
      - name: Check AWX configuration
        shell: powershell
        run: |
          Write-Host "Checking AWX configuration..."

          if ([string]::IsNullOrWhiteSpace("${{ secrets.AWX_HOST }}")) {
              Write-Host "ERROR: AWX_HOST is empty or not configured."
              exit 1
          }

          if ([string]::IsNullOrWhiteSpace("${{ secrets.AWX_TOKEN }}")) {
              Write-Host "ERROR: AWX_TOKEN is empty or not configured."
              exit 1
          }

          if ([string]::IsNullOrWhiteSpace("${{ secrets.AWX_PROJECT_ID }}")) {
              Write-Host "ERROR: AWX_PROJECT_ID is empty or not configured."
              exit 1
          }

          if ([string]::IsNullOrWhiteSpace("${{ secrets.AWX_JOB_TEMPLATE_ID }}")) {
              Write-Host "ERROR: AWX_JOB_TEMPLATE_ID is empty or not configured."
              exit 1
          }

          Write-Host "AWX_HOST is configured."
          Write-Host "AWX_PROJECT_ID is configured."
          Write-Host "AWX_JOB_TEMPLATE_ID is configured."
          Write-Host "AWX_TOKEN is configured."

      # ============================================================
      # Step 2: Test AWX Connectivity
      # ============================================================
      - name: Test AWX connectivity
        shell: powershell
        run: |
          Write-Host "Testing DNS resolution for AWX..."

          Resolve-DnsName awx-test.cbo.com

          Write-Host ""
          Write-Host "Testing HTTPS connectivity to AWX..."

          $connection = Test-NetConnection awx-test.cbo.com -Port 443

          if (-not $connection.TcpTestSucceeded) {
              Write-Host "ERROR: Cannot connect to AWX on TCP port 443."
              exit 1
          }

          Write-Host "AWX HTTPS connectivity successful."

          Write-Host ""
          Write-Host "Testing AWX API..."

          $response = curl.exe -k -s `
            --connect-timeout 10 `
            "${{ secrets.AWX_HOST }}/api/v2/ping/"

          Write-Host "AWX API response:"
          Write-Host $response

          if ([string]::IsNullOrWhiteSpace($response)) {
              Write-Host "ERROR: No response received from AWX API."
              exit 1
          }

          Write-Host "AWX API connectivity successful."

      # ============================================================
      # Step 3: Sync AWX Project
      # ============================================================
      - name: Sync AWX Project Source Code
        shell: powershell
        run: |
          Write-Host "Starting AWX project synchronization..."

          $url = "${{ secrets.AWX_HOST }}/api/v2/projects/${{ secrets.AWX_PROJECT_ID }}/update/"

          Write-Host "AWX Project ID: ${{ secrets.AWX_PROJECT_ID }}"

          $response = curl.exe -k -s -w "`nHTTP_STATUS:%{http_code}" `
            -X POST `
            -H "Authorization: Bearer ${{ secrets.AWX_TOKEN }}" `
            -H "Content-Type: application/json" `
            "$url"

          Write-Host "AWX synchronization response:"
          Write-Host $response

          if ($response -match "HTTP_STATUS:2\d\d") {
              Write-Host "AWX project synchronization requested successfully."
          }
          else {
              Write-Host "ERROR: AWX project synchronization failed."
              exit 1
          }

      # ============================================================
      # Step 4: Wait for AWX Project Synchronization
      # ============================================================
      - name: Wait for AWX project synchronization
        shell: powershell
        run: |
          Write-Host "Waiting for AWX project synchronization..."
          Write-Host "Waiting 15 seconds..."

          Start-Sleep -Seconds 15

          Write-Host "AWX project synchronization wait completed."

      # ============================================================
      # Step 5: Launch AWX Job Template
      # ============================================================
      - name: Launch AWX Deployment Job Template
        shell: powershell
        run: |
          Write-Host "Launching AWX deployment job template..."

          $url = "${{ secrets.AWX_HOST }}/api/v2/job_templates/${{ secrets.AWX_JOB_TEMPLATE_ID }}/launch/"

          Write-Host "AWX Job Template ID: ${{ secrets.AWX_JOB_TEMPLATE_ID }}"

          $response = curl.exe -k -s -w "`nHTTP_STATUS:%{http_code}" `
            -X POST `
            -H "Authorization: Bearer ${{ secrets.AWX_TOKEN }}" `
            -H "Content-Type: application/json" `
            "$url"

          Write-Host "AWX launch response:"
          Write-Host $response

          if ($response -match "HTTP_STATUS:2\d\d") {
              Write-Host "AWX deployment job launched successfully."
          }
          else {
              Write-Host "ERROR: Failed to launch AWX Job Template."
              exit 1
          }



How above steps do(CI/CD)
-----------------------------
Phase 1: You Make a Change (Your Computer)
-------------------------------------------
1. The Code Update: You modify an Ansible playbook or a configuration file in your local repository.
2. The Trigger:      You run git push origin main. Your local changes are uploaded to your GitHub repository.

Phase 2: GitHub Actions Wakes Up (The GitHub Cloud)
--------------------------------------------------
 3. Reading the Map: GitHub detects the new commit on the main branch. It looks inside .github/workflows/ and finds your cd.yml file.
 4. Booting the Runner: GitHub spins up a fresh, clean Ubuntu Linux virtual machine (the GitHub Runner) to execute the steps in your script.

Phase 3: The Project Sync (GitHub ➡️ AWX)
------------------------------------------
 5. The REST Call: The GitHub Runner runs the first curl command. 
It targets your AWX_HOST API, specifically the /projects/ID/update/ 
endpoint. It signs this message securely using your hidden AWX_TOKEN.

6. AWX Pulls the Code: Your AWX server receives this request. 
AWX says, "Ah, GitHub just told me there is new code."
AWX goes to your GitHub repository and runs a git pull in its own background directory.

7. The Intermission: Back in GitHub, the runner reaches the sleep 10 step. It waits for 10 seconds to give AWX enough time to finish downloading your updated Ansible playbooks.

Phase 4: Triggering the Job (GitHub ➡️ AWX)
---------------------------------------------
 8. The Launch Request: The GitHub Runner executes the second curl command targeting the /job_templates/ID/launch/ endpoint.
 9. Passing Variables: In this call, GitHub dynamically includes data: {"extra_vars": {"git_commit_sha": "..."}}. This passes the exact unique identifier of your current Git commit to AWX as a variable.
 10. GitHub Finishes: Because GitHub's job was just to tell AWX to start, the GitHub runner completes its script, turns green, and shuts down.

Phase 5: Execution (AWX ➡️ Your Target Servers)
-----------------------------------------------
 11. Running the Playbook: AWX takes the newly downloaded playbooks from Phase 3 and begins running the specified Job Template.
 12. Infrastructure Changes: The Ansible tasks execute on your target nodes (e.g., configuring Nginx, updating app versions, setting up firewalls)
 13. Final Status: AWX logs the entire playbook execution inside its own dashboard UI so you can see if the configuration succeeded or encountered errors.

 Sample project : Web Server Setup on server 10.1.177.205
----------------------------------------------------------
This playbook automatically installs and starts an Nginx web server on your target machines.
It is a perfect proof-of-concept to see if your CD pipeline is executing tasks correctly.

1. Create a file named deploy-webserver.yml

---
- name: Deploy and Configure Nginx Web Server
  hosts: webservers
  become: true # Runs commands as root/admin

  tasks:
    - name: Ensure Nginx is installed (Ubuntu/Debian)
      apt:
        name: nginx
        state: present
        update_cache: yes
      when: ansible_os_family == "Debian"

    - name: Ensure Nginx is started and enabled
      service:
        name: nginx
        state: started
        enabled: yes

    - name: Deploy custom index.html landing page
      copy:
        dest: /var/www/html/index.html
        content: |
          <h1>🚀 Hello World! Deployed automatically via GitHub & AWX!</h1>
          <p>Commit Hash Used: {{ git_commit_sha | default('Manual Run') }}</p>
        mode: '0644'
